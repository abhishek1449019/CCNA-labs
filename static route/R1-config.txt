!=========================================
! R1 (Cisco 2911) - Static Routing
!=========================================
enable
configure terminal
hostname R1
no ip domain-lookup

! LAN towards PC0
interface GigabitEthernet0/0
 description LAN to PC0 (10.0.0.0/8)
 ip address 10.0.0.1 255.0.0.0
 no shutdown
 exit

! Link to R2
interface GigabitEthernet0/1
 description Link to R2 (20.0.0.0/8)
 ip address 20.0.0.1 255.0.0.0
 no shutdown
 exit

! Static routes (next hop = R2 Gi0/0)
ip route 30.0.0.0 255.0.0.0 20.0.0.2
ip route 40.0.0.0 255.0.0.0 20.0.0.2

end
write memory
